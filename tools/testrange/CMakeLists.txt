cmake_minimum_required(VERSION 3.24)
project(BC2TestRange C CXX)
# App-local injection must not depend on a separately installed VC runtime.
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
  message(FATAL_ERROR "BC2TestRange targets the 32-bit game; configure with an x86 toolchain")
endif()
set(CMAKE_CXX_STANDARD 20)
# The repo's MinHook, for the Give.* detours.
set(MINHOOK_DIR ${CMAKE_CURRENT_SOURCE_DIR}/../../third_party/minhook-1.3.4)
add_library(BC2TestRangeMinHook STATIC ${MINHOOK_DIR}/src/buffer.c ${MINHOOK_DIR}/src/hook.c ${MINHOOK_DIR}/src/trampoline.c ${MINHOOK_DIR}/src/hde/hde32.c)
target_include_directories(BC2TestRangeMinHook PUBLIC ${MINHOOK_DIR}/include)
add_library(BC2TestRange SHARED BC2TestRange.cpp)
target_link_libraries(BC2TestRange PRIVATE BC2TestRangeMinHook)
target_compile_definitions(BC2TestRange PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX _CRT_SECURE_NO_WARNINGS)
target_compile_options(BC2TestRange PRIVATE /W4 /EHa)
add_executable(BC2TestRangeInject BC2TestRangeInject.cpp)
target_compile_definitions(BC2TestRangeInject PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX _CRT_SECURE_NO_WARNINGS UNICODE _UNICODE)
target_compile_options(BC2TestRangeInject PRIVATE /W4)
